ACL VLAN 10

ip access-list extended ACL_VLAN10_GERENCIA
permit udp any any eq bootps
permit udp any any eq bootpc
permit ip 192.168.10.0 0.0.0.255 host 10.0.14.106
permit tcp 192.168.10.0 0.0.0.255 host 10.0.15.102 eq 80
permit tcp 192.168.10.0 0.0.0.255 host 10.0.15.102 eq 443
permit ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
permit ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
permit ip 192.168.10.0 0.0.0.255 192.168.40.0 0.0.0.255

deny ip 192.168.10.0 0.0.0.255 192.168.90.0 0.0.0.255
deny ip 192.168.10.0 0.0.0.255 10.0.0.0 0.255.255.255
